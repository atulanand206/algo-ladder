### u-powmod
--java
static long powMod(long b, long e, long m) {
    long r = 1; b %= m;
    for (; e > 0; e >>= 1, b = b * b % m)
        if ((e & 1) == 1) r = r * b % m;
    return r;
}
--py

--cpp
long long powMod(long long b, long long e, long long m) {
    long long r = 1; b %= m;
    for (; e > 0; e >>= 1, b = b * b % m)
        if (e & 1) r = r * b % m;
    return r;
}
