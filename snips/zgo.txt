func bfs(adj [][]int, s int) []int { // distance from s, -1 if unreachable
    dist := make([]int, len(adj))
    for i := range dist {
        dist[i] = -1
    }
    dist[s] = 0
    q := []int{s}
    for len(q) > 0 {
        v := q[0]
        q = q[1:]
        for _, w := range adj[v] {
            if dist[w] < 0 {
                dist[w] = dist[v] + 1
                q = append(q, w)
            }
        }
    }
    return dist
}

func buildLift(parent []int, LOG int) [][]int { // up[k][v] = 2^k-th ancestor of v
    n := len(parent)
    up := make([][]int, LOG)
    up[0] = append([]int(nil), parent...)
    for k := 1; k < LOG; k++ {
        up[k] = make([]int, n)
        for v := 0; v < n; v++ {
            up[k][v] = up[k-1][up[k-1][v]]
        }
    }
    return up
}

    s := append([]int(nil), a...)
    sort.Ints(s)
    k := 0
    for i, x := range s {
        if i == 0 || x != s[i-1] {
            s[k] = x
            k++
        }
    }
    s = s[:k]

        out[i] = sort.SearchInts(s, x)
    }
    return out

func buildFactorials(n, M int) ([]int, []int) { // fact, invFact
    f, inv := make([]int, n+1), make([]int, n+1)
    f[0] = 1
    for i := 1; i <= n; i++ {
        f[i] = f[i-1] * i % M
    }
    inv[n] = powMod(f[n], M-2, M)
    for i := n; i > 0; i-- {
        inv[i-1] = inv[i] * i % M
    }
    return f, inv
}

func (hp *MinHeap) pop() int { // method of MinHeap above
    top := hp.h[0]
    last := len(hp.h) - 1
    hp.h[0] = hp.h[last]
    hp.h = hp.h[:last]
    for i := 0; ; {
        c := 2*i + 1
        if c >= len(hp.h) {
            break
        }
        if c+1 < len(hp.h) && hp.h[c+1] < hp.h[c] {
            c++
        }
        if hp.h[i] <= hp.h[c] {
            break
        }

        i = c
    }
    return top
}

// uses package-level state: var depth []int; var up [][]int; var LOG int
func lca(u, v int) int {
    if depth[u] < depth[v] {
        u, v = v, u
    }
    for k := LOG - 1; k >= 0; k-- {
        if depth[u]-(1<<k) >= depth[v] {
            u = up[k][u]
        }
    }
    if u == v {
        return u
    }
    for k := LOG - 1; k >= 0; k-- {
        if up[k][u] != up[k][v] {
            u, v = up[k][u], up[k][v]
        }
    }
    return up[0][u]
}

        } else {

            b = b.next

// needs container/list
type LRU struct {
    cap   int
    order *list.List // front = most recently used; elements hold [2]int{key, value}
    pos   map[int]*list.Element
}

func newLRU(c int) *LRU { return &LRU{cap: c, order: list.New(), pos: map[int]*list.Element{}} }
func (l *LRU) get(k int) int {
    e, ok := l.pos[k]
    if !ok {
        return -1
    }
    l.order.MoveToFront(e)
    return e.Value.([2]int)[1]
}
func (l *LRU) put(k, v int) {
    if e, ok := l.pos[k]; ok {
        l.order.Remove(e)
    }
    l.pos[k] = l.order.PushFront([2]int{k, v})
    if l.order.Len() > l.cap {
        last := l.order.Back()
        delete(l.pos, last.Value.([2]int)[0])
        l.order.Remove(last)
    }
}

func matMul(a, b [][]int, M int) [][]int {
    n := len(a)
    c := make([][]int, n)
    for i := range c {
        c[i] = make([]int, n)
    }
    for i := 0; i < n; i++ {
        for k := 0; k < n; k++ {
            for j := 0; j < n; j++ {
                c[i][j] = (c[i][j] + a[i][k]*b[k][j]) % M
            }
        }
    }
    return c
}

func matPow(base [][]int, e, M int) [][]int {
    n := len(base)
    res := make([][]int, n)
    for i := range res {
        res[i] = make([]int, n)
        res[i][i] = 1 // identity
    }
    for ; e > 0; e >>= 1 {
        if e&1 == 1 {
            res = matMul(res, base, M)
        }
        base = matMul(base, base, M)
    }
    return res

func build2d(a [][]int) [][]int { // p[i][j] = sum of the top-left i x j block
    R, C := len(a), len(a[0])
    p := make([][]int, R+1)
    for i := range p {
        p[i] = make([]int, C+1)
    }
    for i := 1; i <= R; i++ {
        for j := 1; j <= C; j++ {
            p[i][j] = a[i-1][j-1] + p[i-1][j] + p[i][j-1] - p[i-1][j-1]
        }
    }
    return p
}

func phi(n int) int { // how many numbers in 1..n share no factor with n
    r := n
    for d := 2; d*d <= n; d++ {
        if n%d == 0 {
            for n%d == 0 {
                n /= d
            }
            r -= r / d
        }
    }
    if n > 1 {
        r -= r / n

    h = h[:last]
    for i := 0; ; {
        c := 2*i + 1
        if c >= len(h) {
            break
        }

            c++
        }
        if h[i][0] <= h[c][0] {
            break
        }
        h[i], h[c] = h[c], h[i]
        i = c
    }
    *pq = h
    return top
}

func topoOrder(adj [][]int) []int { // shorter than n means a cycle
    n := len(adj)
    indeg := make([]int, n)
    for v := 0; v < n; v++ {
        for _, w := range adj[v] {
            indeg[w]++
        }
    }
    var q, order []int
    for v := 0; v < n; v++ {
        if indeg[v] == 0 {
            q = append(q, v)
        }
    }
    for len(q) > 0 {
        v := q[0]
        q = q[1:]
        order = append(order, v)
        for _, w := range adj[v] {
            indeg[w]--
            if indeg[w] == 0 {
                q = append(q, w)
            }
        }
    }
    return order
}

type Trie struct {
    nxt [][26]int // nxt[v][c] = child of node v by letter c, 0 = none
    end []bool
}

func newTrie() *Trie { return &Trie{nxt: make([][26]int, 1), end: make([]bool, 1)} }
func (t *Trie) insert(w string) {
    v := 0
    for i := 0; i < len(w); i++ {
        c := w[i] - 'a'
        if t.nxt[v][c] == 0 {
            t.nxt[v][c] = len(t.nxt)
            t.nxt = append(t.nxt, [26]int{})
            t.end = append(t.end, false)
        }
        v = t.nxt[v][c]
    }
    t.end[v] = true
}
func (t *Trie) search(w string) bool {
    v := 0
    for i := 0; i < len(w); i++ {
        v = t.nxt[v][w[i]-'a']
        if v == 0 {
            return false
        }
    }
    return t.end[v]
}

func longestWindow(a []int, k int) int { // longest stretch with at most k distinct values
    cnt := map[int]int{}
    best := 0
    for l, r := 0, 0; r < len(a); r++ {
        cnt[a[r]]++ // add the new item
        for len(cnt) > k { // drop old ones
            cnt[a[l]]--
            if cnt[a[l]] == 0 {
                delete(cnt, a[l])
            }
            l++
        }
        best = max(best, r-l+1)
    }
    return best
}
### u-zfn

func zFunction(s string) []int { // z[i] = how far s matches its own start from i
    n := len(s)
    z := make([]int, n)
    for i, l, r := 1, 0, 0; i < n; i++ {
        if i < r {
            z[i] = min(r-i, z[i-l])
        }
        for i+z[i] < n && s[z[i]] == s[i+z[i]] {
            z[i]++
        }
        if i+z[i] > r {
            l, r = i, i+z[i]
        }
    }
    return z
}
