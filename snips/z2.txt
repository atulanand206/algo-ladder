    int n = adj.size();

    Deque<Integer> q = new ArrayDeque<>();

    while (!q.isEmpty()) {
        int v = q.poll(); order.add(v);
        for (int w : adj.get(v)) if (--indeg[w] == 0) q.add(w);
    }
    return order;
}
